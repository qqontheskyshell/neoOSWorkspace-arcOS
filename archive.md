                                            - Q_e_imag: imaginary part of coupling Q-factor

                                                """

# LoRaBLEChat

  

SwiftUI iOS starter app that uses CoreBluetooth to connect to a BLE peripheral (such as a LoRa-to-BLE bridge), provides a small chat UI, and sends/receives UTF-8 messages over BLE characteristics.

  

## What this app does

- Scans for a BLE peripheral advertising a service UUID you set in BLEDefinitions.swift

- Connects with CoreBluetooth

- Discovers TX/RX characteristics

- Sends UTF-8 text to the peripheral with writeValue

- Receives UTF-8 text through characteristic notifications

  

## Before building

1. Open the generated folder in Xcode.

2. Create a new **iOS App** project named LoRaBLEChat with SwiftUI.

3. Replace the generated Swift files with the files in this folder, or drag these files into the app target.

4. In Info.plist, add:

   - Privacy - Bluetooth Always Usage Description

   - Privacy - Bluetooth Peripheral Usage Description

1. Set the UUIDs in BLEDefinitions.swift to match your LoRa BLE bridge.

2. Build on a real iPhone; CoreBluetooth scanning is not supported in the iOS simulator.

  

## Typical BLE mapping

- Service UUID: custom UART-style service from your bridge

- TX characteristic: iPhone writes outgoing chat text here

- RX characteristic: bridge notifies incoming LoRa text here

  

## Notes

This is app-level BLE communication. It does not integrate with iOS CommCenter or create a native cellular/network interface.

import Foundation

import CoreBluetooth

  

final class BLEChatViewModel: NSObject, ObservableObject {

    @Published var isBluetoothReady = false

    @Published var isScanning = false

    @Published var isConnected = false

    @Published var peripheralName = "Not connected"

    @Published var messages: [ChatMessage] = []

    @Published var discoveredPeripherals: [CBPeripheral] = []

    @Published var statusText = "Bluetooth starting..."

  

    private var centralManager: CBCentralManager!

    private var targetPeripheral: CBPeripheral?

    private var txCharacteristic: CBCharacteristic?

    private var rxCharacteristic: CBCharacteristic?

    private var discoveredIDs = Set<UUID>()

  

    override init() {

        super.init()

        centralManager = CBCentralManager(delegate: self, queue: .main)

    }

  

    func startScan() {

        guard centralManager.state == .poweredOn else {

            statusText = "Bluetooth is unavailable."

            return

        }

        discoveredPeripherals.removeAll()

        discoveredIDs.removeAll()

        isScanning = true

        statusText = "Scanning for BLE LoRa bridge..."

        centralManager.scanForPeripherals(withServices: [BLEDefinitions.serviceUUID], options: [CBCentralManagerScanOptionAllowDuplicatesKey: false])

    }

  

    func stopScan() {

        centralManager.stopScan()

        isScanning = false

        statusText = isConnected ? "Connected" : "Scan stopped"

    }

  

    func connect(_ peripheral: CBPeripheral) {

        stopScan()

        targetPeripheral = peripheral

        peripheral.delegate = self

        peripheralName = peripheral.name ?? "Unnamed device"

        statusText = "Connecting to \(peripheralName)..."

        centralManager.connect(peripheral, options: nil)

    }

  

    func disconnect() {

        guard let targetPeripheral else { return }

        centralManager.cancelPeripheralConnection(targetPeripheral)

    }

  

    func send(_ text: String) {

        let trimmed = text.trimmingCharacters(in: .whitespacesAndNewlines)

        guard !trimmed.isEmpty else { return }

        guard let targetPeripheral, let txCharacteristic else {

            statusText = "Not ready to send"

            return

        }

        guard let data = trimmed.data(using: .utf8) else {

            statusText = "UTF-8 encoding failed"

            return

        }

        let writeType: CBCharacteristicWriteType = txCharacteristic.properties.contains(.writeWithoutResponse) ? .withoutResponse : .withResponse

        targetPeripheral.writeValue(data, for: txCharacteristic, type: writeType)

        messages.append(ChatMessage(text: trimmed, isLocal: true, timestamp: Date()))

    }

  

    private func appendIncoming(_ data: Data) {

        guard let text = String(data: data, encoding: .utf8) else {

            statusText = "Received non-UTF8 payload (\(data.count) bytes)"

            return

        }

        messages.append(ChatMessage(text: text, isLocal: false, timestamp: Date()))

    }

}

  

extension BLEChatViewModel: CBCentralManagerDelegate {

    func centralManagerDidUpdateState(_ central: CBCentralManager) {

        switch central.state {

        case .poweredOn:

            isBluetoothReady = true

            statusText = "Bluetooth ready"

        case .poweredOff:

            isBluetoothReady = false

            statusText = "Bluetooth powered off"

        case .unauthorized:

            isBluetoothReady = false

            statusText = "Bluetooth unauthorized"

        case .unsupported:

            isBluetoothReady = false

            statusText = "Bluetooth unsupported"

        case .resetting:

            isBluetoothReady = false

            statusText = "Bluetooth resetting"

        case .unknown:

            fallthrough

        @unknown default:

            isBluetoothReady = false

            statusText = "Bluetooth state unknown"

        }

    }

  

    func centralManager(_ central: CBCentralManager, didDiscover peripheral: CBPeripheral, advertisementData: [String : Any], rssi RSSI: NSNumber) {

        guard !discoveredIDs.contains(peripheral.identifier) else { return }

        discoveredIDs.insert(peripheral.identifier)

        discoveredPeripherals.append(peripheral)

    }

  

    func centralManager(_ central: CBCentralManager, didConnect peripheral: CBPeripheral) {

        isConnected = true

        peripheralName = peripheral.name ?? "Unnamed device"

        statusText = "Connected to \(peripheralName)"

        peripheral.discoverServices([BLEDefinitions.serviceUUID])

    }

  

    func centralManager(_ central: CBCentralManager, didFailToConnect peripheral: CBPeripheral, error: Error?) {

        isConnected = false

        statusText = "Connect failed: \(error?.localizedDescription ?? "unknown error")"

    }

  

    func centralManager(_ central: CBCentralManager, didDisconnectPeripheral peripheral: CBPeripheral, error: Error?) {

        isConnected = false

        txCharacteristic = nil

        rxCharacteristic = nil

        statusText = error == nil ? "Disconnected" : "Disconnected: \(error!.localizedDescription)"

    }

}

  

extension BLEChatViewModel: CBPeripheralDelegate {

    func peripheral(_ peripheral: CBPeripheral, didDiscoverServices error: Error?) {

        if let error {

            statusText = "Service discovery failed: \(error.localizedDescription)"

            return

        }

        guard let services = peripheral.services else { return }

        for service in services where service.uuid == BLEDefinitions.serviceUUID {

            peripheral.discoverCharacteristics([BLEDefinitions.txUUID, BLEDefinitions.rxUUID], for: service)

        }

    }

  

    func peripheral(_ peripheral: CBPeripheral, didDiscoverCharacteristicsFor service: CBService, error: Error?) {

        if let error {

            statusText = "Characteristic discovery failed: \(error.localizedDescription)"

            return

        }

        guard let characteristics = service.characteristics else { return }

        for characteristic in characteristics {

            if characteristic.uuid == BLEDefinitions.txUUID {

                txCharacteristic = characteristic

            } else if characteristic.uuid == BLEDefinitions.rxUUID {

                rxCharacteristic = characteristic

                peripheral.setNotifyValue(true, for: characteristic)

            }

        }

        if txCharacteristic != nil && rxCharacteristic != nil {

            statusText = "Ready to chat"

        }

    }

  

    func peripheral(_ peripheral: CBPeripheral, didUpdateValueFor characteristic: CBCharacteristic, error: Error?) {

        if let error {

            statusText = "Receive failed: \(error.localizedDescription)"

            return

        }

        guard characteristic.uuid == BLEDefinitions.rxUUID,

              let value = characteristic.value else { return }

        appendIncoming(value)

    }

  

    func peripheral(_ peripheral: CBPeripheral, didWriteValueFor characteristic: CBCharacteristic, error: Error?) {

        if let error {

            statusText = "Send failed: \(error.localizedDescription)"

        }

    }

}
