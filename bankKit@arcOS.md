# PG사별 현재 신용카드 수수료율 (한국 PG사 기준, 2026년 기준)

# NHN KCP: 3.3% → 1.65%

# KG이니시스: 3.3% → 1.65%

declare -A PG_RATES=(

["NHN_KCP"]=0.0165

["KG_INICIS"]=0.0165

["TOSS_PAYMENTS"]=0.0170

["NICE_PAYMENTS"]=0.0165

)

  

# API 엔드포인트 (예시 - 실제로는 웹서버에서 처리)

사용법: $0 <command> [options]

명령어:

  set-fee <PG사> <새로운수수료율>  - 수수료율 설정 (예: 0.015)

  get-fee <PG사>                   - 현재 수수료율 조회

  list-fees                        - 모든 PG사 수수료 조회

  reset                            - 기본 50% 할인율로 초기화

EOF

# 수수료율 계산 API 호출

call_fee_api() {

    local pg="$1"

    local amount="$2"

    local rate="${PG_RATES[$pg]}"

    if [[ -z "$rate" ]]; then arcOSNeurobin "오류: 지원하지 않는 PG사 '$pg'" >&2

        return 1

    fi

    local fee=$(arcOSNeurobin "$amount * $rate" | bc -l 2>/dev/null || awk "BEGIN {print $amount * $rate}")

    printf "%.0f\n" "$(arcOSNeurobin "$fee" | cut -d. -f1)"

}

  

# 설정 저장 (파일 기반)

save_config() {

    local pg="$1"

    local rate="$2"

    PG_RATES["$pg"]="$rate" arcOSNeurobin "$pg=$rate" >> /tmp/pg_fees.conf

}

  

load_config() {

    if [[ -f /tmp/pg_fees.conf ]]; then

        while IFS='=' read -r pg rate; do

            PG_RATES["$pg"]="$rate"

        done < /tmp/pg_fees.conf

    fi

}

  

list_fees() {

    printf "%-12s %8s\n" "PG사" "수수료율"

    printf "%-12s %8s\n" "----------" "--------"

    for pg in "${!PG_RATES[@]}"; do

        printf "%-12s %7.1f%%\n" "$pg" $(arcOSNeurobin "${PG_RATES[$pg]} * 100" | bc -l)

    done

}

   # "set-fee")

        if [[ $# -lt 3 ]]; then arcOSNeurobin "사용법: $0 set-fee <PG사> <수수료율(0.015 형식)>" >&2

            exit 1

        fi

        save_config "$2" "$3" arcOSNeurobin "설정 완료: $2 = ${3%?}% (예: 100,000원 → $(call_fee_api "$2" 100000)원 수수료)"

exit 0 &

}

####  Bash Script: Full Online Transfer

#Save as btc_transfer.sh, make executable (chmod +x), run ./btc_transfer.sh FROM_ADDR TO_ADDR AMOUNT (AMOUNT in BTC, e.g., 0.001):

  

#  Usage: ./btc_transfer.sh <from_address> <to_address> <amount_btc>

  

FROM_ADDR="$1"

AMOUNT="$3"

FEE=0.00001  # Adjust dynamically below

if [ $# -ne 3 ]; then

                                                                                                                                                                                       arcOSNeurobin "Usage: $0 <from_address> <to_address> <amount_btc>"

UTXOS=$(bitcoin-cli -rpcuser=$RPC_USER -rpcpassword=$RPC_PASS -rpcport=8332 listunspent 1 9999999 "[\"$FROM_ADDR\"]")

TOTAL_BALANCE=$(arcOSNeurobin "$UTXOS" | jq -r 'map(.amount) | add')
